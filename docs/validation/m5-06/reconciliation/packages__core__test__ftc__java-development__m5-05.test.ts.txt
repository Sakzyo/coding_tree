  50 // This fixture consumes exactly the declared root files, excluding only its build/cache directories.
  51 // It is not a Gradle inventory or an OS adapter. Every fixture mutation is serialized and increments
  52 // the trusted complete interval generation; reading endpoint hashes alone cannot establish that interval.
  53 async function fixture(root: string) {
  54   const project = context(root)
  55   const toolchain = descriptor(root)
  56   const inventory = new Set([
  57     "Main.java",
  58     "Unopened.java",
  59     "gradlew",
  60     "build.gradle",
  61     "settings.gradle",
  62     "resource.xml",
  63     "configuration.json",
  64   ])
  65   await Promise.all(
  66     [...inventory].map((name) => Bun.write(path.join(root, name), name === "Main.java" ? "saved" : `fixture ${name}`)),
  67   )
  68   const started = gate()
  69   const acquired = gate()
  70   const finish = gate()
  71   const outputFinish = gate()
  72   const firstOutput = gate()
  73   const processCleanup = gate()
  74   const leaseCleanup = gate()
  75   const state = {
  76     generation: 0,
  77     configurationRevision: "config-1",
  78     authorized: project,
  79     acquisitions: 0,
  80     starts: 0,
  81     processesClosed: 0,
  82     leasesClosed: 0,
  83     cancellations: 0,
  84     revalidations: 0,
  85     acquisition: Effect.void as Effect.Effect<void, FtcJava.BuildError>,
  86     start: Effect.void as Effect.Effect<void, FtcJava.BuildError>,
  87     processRelease: Effect.void as Effect.Effect<void, FtcJava.BuildError>,
  88     leaseRelease: Effect.void as Effect.Effect<void, FtcJava.BuildError>,
  89     revalidation: Effect.void as Effect.Effect<void, FtcJava.BuildError>,
  90     exclusions: Effect.succeed([]) as Effect.Effect<readonly FtcJava.BuildExclusion[]>,
  91     termination: { exitCode: 0 } as JavaBuild.Termination,
  92     exitFailure: Effect.void as Effect.Effect<void, FtcJava.BuildError>,
  93     basisMutation: (_basis: JavaBuild.SavedInputBasis) => {},
  94     validationMutation: (_validation: JavaBuild.InputValidation) => {},
  95     log: { sequence: 0, stream: "stdout" as const, text: "actual stdout\n" },
  96     output: undefined as Stream.Stream<FtcJava.BuildLog, FtcJava.BuildError> | undefined,
  97     seen: [] as { lease: JavaBuild.SavedInputLease; toolchain: FtcEnvironment.ToolchainDescriptor }[],
  98   }
  99   const snapshot = () =>
 100     Effect.promise(async () => {
 101       const entries = await Promise.all(
 102         [...inventory].sort().map(async (name) => [name, await Bun.file(path.join(root, name)).text()] as const),
 103       )
 104       return {
 105         bytes: Object.fromEntries(entries),
 106         sourceRevision: createHash("sha256").update(JSON.stringify(entries)).digest("hex"),
 107       }
 108     })
 109   const validation = () =>
 110     Effect.gen(function* () {
 111       const saved = yield* snapshot()
 112       const value = {
 113         project: state.authorized,
 114         toolchain,
 115         sourceRevision: saved.sourceRevision,
 116         configurationRevision: state.configurationRevision,
 117         generation: String(state.generation),
 118         exclusions: yield* state.exclusions,
 119       }
 120       state.validationMutation(value)
 121       return value
 122     })
 123   const ports: JavaBuild.Ports = {
 124     projects: { resolve: () => Effect.succeed(state.authorized) },
 125     inputs: {
 126       acquire: (input) =>
 127         Effect.gen(function* () {
 128           state.acquisitions++
 129           expect(input.project).toEqual(project)
 130           expect(input.toolchain).toEqual(toolchain)
 131           const saved = yield* snapshot()
 132           const basis: JavaBuild.SavedInputBasis = {
 133             project,
 134             toolchain,
 135             sourceRevision: saved.sourceRevision,
 136             configurationRevision: state.configurationRevision,
 137             generation: String(state.generation),
 138             scope: "complete",
 139             mode: "immutable",
 140             recipe: {
 141               identity: "owned-complete-fixture",
 142               wrapper: toolchain.gradleWrapper,
 143               buildJdk: toolchain.buildJdk,
 144               androidSdk: toolchain.androidSdk,
 145             },
 146             exclusions: yield* state.exclusions,
 147           }
 148           state.basisMutation(basis)
 149           const released: { observation?: JavaBuild.InputValidation } = {}
 150           const release = Effect.gen(function* () {
 151             if (released.observation) return released.observation
 152             yield* open(leaseCleanup)
 153             yield* state.leaseRelease
 154             released.observation = yield* validation()
 155             state.leasesClosed++
 156             return released.observation
 157           })
 158           yield* Effect.addFinalizer(() => release.pipe(Effect.orDie, Effect.asVoid))
 159           yield* open(acquired)
 160           yield* state.acquisition
 161           return {
 162             basis,
 163             execution: Object.freeze({ bytes: Object.freeze(saved.bytes) }),
 164             release,
 165             revalidate: () =>
 166               Effect.gen(function* () {
 167                 state.revalidations++
 168                 yield* state.revalidation
 169                 return yield* validation()
 170               }),
 171           }
 172         }),
 173     },
 174     process: {
 175       start: (input) =>
 176         Effect.gen(function* () {
 177           state.starts++
 178           state.seen.push(input)
 179           yield* state.start
 180           const closed = { value: false }
 181           const release = Effect.gen(function* () {
 182             if (closed.value) return
 183             yield* open(processCleanup)
 184             yield* state.processRelease
 185             closed.value = true
 186             state.processesClosed++
 187           })
 188           yield* Effect.addFinalizer(() => release.pipe(Effect.orDie))
 189           yield* open(started)
 190           return {
 191             output:
 192               state.output ??
 193               Stream.make(state.log).pipe(
 194                 Stream.tap(() => open(firstOutput)),
 195                 Stream.concat(
 196                   Stream.fromEffect(
 197                     Deferred.await(outputFinish).pipe(
 198                       Effect.as({ sequence: 1, stream: "stderr" as const, text: "actual stderr\n" }),
 199                     ),
 200                   ),
 201                 ),
 202               ),
 203             termination: Deferred.await(finish).pipe(
 204               Effect.andThen(state.exitFailure),
 205               Effect.map(() => state.termination),
 206             ),
 207             cancelAndJoin: Effect.gen(function* () {
 208               state.cancellations++
 209               yield* open(finish)
 210               yield* open(outputFinish)
 211               return state.termination
 212             }),
 213             release,
 214           }
 215         }),
 216     },
 217   }
 218   const request = { project, toolchain, configurationRevision: "config-1" }
 219   const complete = open(finish).pipe(Effect.andThen(open(outputFinish)))
 220   const write = (name: string, bytes: string) =>
 221     Effect.promise(async () => {
 222       inventory.add(name)
 223       await Bun.write(path.join(root, name), bytes)
 224       state.generation++
 225     })
 226   return {
 227     project,
 228     toolchain,
 229     state,
 230     ports,
 231     request,
 232     inventory,
 233     write,
 234     snapshot,
 235     validation,
 236     started,
 237     acquired,
 238     finish,
 239     outputFinish,
 240     firstOutput,
 241     processCleanup,
 242     leaseCleanup,
 243     complete,
 244   }
 245 }

 247 async function isolated<A, E>(
 248   body: (owner: JavaBuild.Interface, supplied: Awaited<ReturnType<typeof fixture>>) => Effect.Effect<A, E, Scope.Scope>,
 249 ) {
 250   await using tmp = await tmpdir()
 251   const supplied = await fixture(tmp.path)
 252   return await run(
 253     Effect.gen(function* () {
 254       const owner = yield* JavaBuild.make(supplied.ports)
 255       return yield* body(owner, supplied)
 256     }),
 257   )
