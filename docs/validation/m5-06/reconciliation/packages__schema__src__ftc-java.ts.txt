   3 import { Schema } from "effect"
   4 import { Project } from "./project"
   5 import { FtcProject } from "./ftc-project"
   6 import { FtcEnvironment } from "./ftc-environment"
   7 import { AbsolutePath, NonNegativeInt, optional, statics } from "./schema"

  10 export const DocumentID = Schema.String.check(Schema.isStartsWith("doc_"))
  11   .pipe(Schema.brand("FtcJava.DocumentID"))
  12   .annotate({ identifier: "FtcJava.DocumentID" })
  13   .pipe(statics((schema) => ({ create: () => schema.make(`doc_${ascending()}`) })))
  14 export type DocumentID = typeof DocumentID.Type
  15 
  16 export interface DocumentSnapshot extends Schema.Schema.Type<typeof DocumentSnapshot> {}
  17 export const DocumentSnapshot = Schema.Struct({
  18   documentID: DocumentID,
  19   projectID: Project.ID,
  20   path: AbsolutePath.check(Schema.isPattern(/^(?:\/|[a-zA-Z]:[\\/]|\\\\)/)),
  21   bufferRevision: NonNegativeInt,
  22   // Revision of the most recently observed disk bytes; buffer revisions are independent.
  23   diskRevision: NonNegativeInt,
  24   text: Schema.String,
  25   dirty: Schema.Boolean,
  26 }).annotate({ identifier: "FtcJava.DocumentSnapshot" })
  27 
  28 export interface DocumentEvent extends Schema.Schema.Type<typeof DocumentEvent> {}
  29 export const DocumentEvent = Schema.Struct({
  30   type: Schema.Literals(["opened", "changed", "saved", "disk_changed"]),
  31   snapshot: DocumentSnapshot,
  32 }).annotate({ identifier: "FtcJava.DocumentEvent" })
  33 
  34 export interface Revision extends Schema.Schema.Type<typeof Revision> {}

  38   diskRevision: NonNegativeInt,
  39 }).annotate({ identifier: "FtcJava.Revision" })
  40 
  41 export interface SaveRequest extends Schema.Schema.Type<typeof SaveRequest> {}

  47 
  48 export interface EditProposal extends Schema.Schema.Type<typeof EditProposal> {}
  49 export const EditProposal = Schema.Struct({
  50   projectID: Project.ID,
  51   edits: Schema.Array(
  52     Schema.Struct({ path: Schema.String, expectedRevision: Revision, replacement: Schema.String }),
  53   ).check(Schema.isMinLength(1)),
  54   explanation: Schema.String,
  55 }).annotate({ identifier: "FtcJava.EditProposal" })
  56 
  57 // Authorization is an opaque runtime capability supplied to Core, never a serializable proposal field.
  58 export interface EditConflict extends Schema.Schema.Type<typeof EditConflict> {}
  59 export const EditConflict = Schema.Struct({
  60   path: Schema.String,
  61   reason: Schema.Literals(["revision", "dirty", "disk_changed"]),
  62   current: DocumentSnapshot,
  63   diskText: Schema.String,
  64   choices: Schema.Array(Schema.Literals(["save", "merge", "defer"])),
  65 }).annotate({ identifier: "FtcJava.EditConflict" })
  66 
  67 export interface DocumentError extends Schema.Schema.Type<typeof DocumentError> {}
  68 export const DocumentError = Schema.Struct({
  69   code: Schema.Literals([
  70     "project_unauthorized",
  71     "path_outside_project",
  72     "file_unavailable",
  73     "document_not_open",
  74     "revision_conflict",
  75     "invalid_document_input",
  76     "owner_closed",
  77     "edit_unauthorized",
  78   ]),
  79   projectID: optional(Project.ID),
  80   path: optional(Schema.String),
  81   expectedRevision: optional(NonNegativeInt),
  82   actualRevision: optional(NonNegativeInt),
  83   conflict: optional(EditConflict),
  84   outcome: optional(Schema.Literals(["unknown"])),
  85 }).annotate({ identifier: "FtcJava.DocumentError" })
  86 
  87 export const EditResult = Schema.Union([
  88   Schema.Struct({
  89     kind: Schema.Literal("applied"),
  90     proposal: EditProposal,
  91     applied: Schema.Array(DocumentSnapshot),
  92   }),
  93   Schema.Struct({
  94     kind: Schema.Literal("conflict"),
  95     proposal: EditProposal,
  96     applied: Schema.Array(DocumentSnapshot),
  97     conflicts: Schema.Array(EditConflict).check(Schema.isMinLength(1)),
  98   }),
  99   Schema.Struct({
 100     kind: Schema.Literal("failed"),
 101     proposal: EditProposal,
 102     applied: Schema.Array(DocumentSnapshot),
 103     error: DocumentError,
 104     uncertainPaths: Schema.Array(Schema.String),
 105   }),
 106 ]).annotate({ identifier: "FtcJava.EditResult" })
 107 export type EditResult = typeof EditResult.Type

 163 // Success describes only the captured saved basis; exclusions never authorize a displayed dirty program.
 164 export interface BuildEvidence extends Schema.Schema.Type<typeof BuildEvidence> {}
 165 export const BuildEvidence = Schema.Struct({
 166   buildID: BuildID,
 167   projectID: Project.ID,
 168   sourceRevision: BuildRevision,
 169   configurationRevision: BuildRevision,
 170   inputChanged: Schema.Boolean,
 171   exitCode: optional(Schema.Int),
 172   signal: optional(BuildRevision),
 173   status: Schema.Literals(["succeeded", "failed", "cancelled", "outdated"]),
 174   logs: Schema.Array(BuildLog),
 175   outputComplete: Schema.Boolean,
 176   savedInputsOnly: Schema.Literal(true),
 177   // Absence of settlement exclusions means final observation was unavailable, never an observed clean state.
 178   exclusions: Schema.Struct({
 179     initial: Schema.Array(BuildExclusion),
 180     settlement: optional(Schema.Array(BuildExclusion)),
 181   }),
 182   errors: Schema.Array(BuildError),
 183 }).annotate({ identifier: "FtcJava.BuildEvidence" })
