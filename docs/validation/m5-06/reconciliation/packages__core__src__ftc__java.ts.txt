  29 export interface Interface extends JavaDocuments.Interface, JavaBuild.Interface {}
  30 export class Service extends Context.Service<Service, Interface>()("@opencode/FtcJava") {}
  31 
  32 // Application wiring supplies authorized canonical project records and the filesystem port.
  33 export const layer = (ports: JavaDocuments.Ports) =>
  34   Layer.effect(
  35     Service,
  36     JavaDocuments.make(ports).pipe(
  37       Effect.map((documents) => ({
  38         ...documents,
  39         build: unavailable,
  40         startBuild: unavailable,
  41         readBuild: unavailable,
  42       })),
  43     ),
  44   )
  45 
  46 const unavailable = () => Effect.fail({ code: "build_unavailable" as const })
  47 
  48 export const layerWithBuilds = (ports: { readonly documents: JavaDocuments.Ports; readonly builds: JavaBuild.Ports }) =>
  49   Layer.effect(
  50     Service,
  51     Effect.gen(function* () {
  52       const documents = yield* JavaDocuments.make(ports.documents)
  53       const builds = yield* JavaBuild.make(ports.builds)
  54       return { ...documents, ...builds }
  55     }),
  56   )
